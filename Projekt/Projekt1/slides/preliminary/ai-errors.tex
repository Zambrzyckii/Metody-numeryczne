  \frametitle{AI skróciło cytat i przedstawiło go jako pełny}
  \begin{description}
    \item[Co się stało] na slajd trafił cytat polecenia dla agenta. Agent
      wyciął z niego fragmenty, ale nie zaznaczył miejsc cięcia. Cytat
      wyglądał na kompletny, choć nim nie był.
    \item[Jak to wykryto] recenzent AI porównał cytat ze slajdu z pełnym
      zapisem polecenia w dzienniku pracy i wymienił brakujące fragmenty.
    \item[Jak naprawiono] każde cięcie oznaczono znakiem [...], a błędny
      opis w dzienniku poprawiono.
  \end{description}
\end{frame}

\begin{frame}
  \frametitle{Poprawka zalecona przez recenzenta AI zepsuła układ slajdu}
  \begin{description}
    \item[Co się stało] recenzent AI zalecił drobną zmianę formatowania
      tekstu. Po zmianie jeden wiersz stał się dłuższy i nasunął się na
      sąsiednią część slajdu.
    \item[Jak to wykryto] w kolejnej recenzji ten sam recenzent zauważył
      nachodzące na siebie elementy w PDF i wskazał, że przyczyną było
      jego wcześniejsze zalecenie. Kompilacja nie zgłosiła żadnego
      ostrzeżenia, błąd był widoczny tylko w PDF.
    \item[Jak naprawiono] układ slajdu zmieniono tak, aby oba elementy
      stały jeden pod drugim.
  \end{description}
\end{frame}
