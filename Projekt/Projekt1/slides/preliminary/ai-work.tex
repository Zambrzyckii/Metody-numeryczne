% Planned AI workflow: agent roles, source hierarchy, checkpoints.

\begin{frame}
  \frametitle{Kod, slajdy i recenzje powstają w czterech rolach agentów Claude Code}
  \begin{itemize}
    \item Narzędzie: Claude Code; sesja główna jest orkiestratorem i deleguje
      prace przez karty zadań.
    \item Role tworzące: coder (Python), latex-assistant (slajdy); role
      recenzujące: code-reviewer i presentation-reviewer, wyłącznie
      z narzędziami odczytu.
    \item Karta zadania zawiera cel, wycinek wymagań z referencjami do
      instrukcji, listę plików i kryteria akceptacji.
    \item Decyzje projektowe i interpretacje wyników pozostają po stronie
      autora.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Jedno źródło prawdy:\\
    od instrukcji prowadzącego, przez brief, do agentów}
  \begin{itemize}
    \item Projekt zaczyna się od pliku wsadowego (brief): cele z tabel
      punktacji, role agentów, konwencje kodu i styl slajdów.
    \item Z briefu wygenerowano CLAUDE.md oraz docs/ z wycinkami wymagań,
      każdy z referencją do strony instrukcji.
    \item Hierarchia źródeł: PDF-y prowadzącego wygrywają z briefem, brief
      z docs/, a docs/ z instrukcjami agentów.
    \item Zmiany płyną wyłącznie w dół hierarchii, nigdy z pominięciem
      briefu.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Decyzje zapadają w pięciu punktach kontrolnych użytkownika}
  \begin{itemize}
    \item Punkty kontrolne: weryfikacja danych, zrozumienie kodu, tezy
      slajdów, kierunek eksperymentu, interpretacje wyników.
    \item Orkiestrator zatrzymuje się na każdym punkcie i nie kontynuuje
      bez decyzji użytkownika.
    \item Interpretacje wyników nie są delegowane agentom.
    \item Dziennik pracy z AI ma dwie warstwy: rejestr decyzji oraz zadania
      agentów.
  \end{itemize}
\end{frame}
